"use server";

import { z } from "zod";
import { revalidatePath } from "next/cache";
import { orderSchema } from "../lib/schema";
import { createOrder, getOrder, markCancelled } from "../lib/db";
import { getSession } from "../lib/session";
import { dishes } from "./menu/dishes";

const placeOrderSchema = orderSchema.extend({
  address: z
    .string()
    .trim()
    .min(5, "Enter your delivery address")
    .max(200, "Address is too long"),
});

export async function placeOrder(prevState, formData) {
  const user = await getSession();
  if (!user) return { error: "Not signed in" };

  const values = {
    name: formData.get("name") ?? "",
    phone: formData.get("phone") ?? "",
    address: formData.get("address") ?? "",
    notes: formData.get("notes") ?? "",
  };

  let requested = [];
  try {
    requested = JSON.parse(String(formData.get("items") ?? "[]"));
  } catch {
    requested = [];
  }

  const items = [];
  if (Array.isArray(requested)) {
    for (const line of requested) {
      const dish = dishes.find((d) => d.id === line?.id);
      const quantity = Number(line?.quantity);
      if (!dish || !Number.isInteger(quantity) || quantity < 1 || quantity > 20) continue;
      items.push({ id: dish.id, name: dish.name, price: dish.price, quantity });
    }
  }

  if (items.length === 0) {
    return { error: "Your cart is empty", values };
  }

  const parsed = placeOrderSchema.safeParse({
    name: values.name,
    phone: values.phone,
    address: values.address,
    notes: values.notes || undefined,
  });

  if (!parsed.success) {
    return {
      error: "Validation failed",
      fieldErrors: parsed.error.flatten().fieldErrors,
      values,
    };
  }

  const total = items.reduce((sum, item) => sum + item.price * item.quantity, 0);

  const order = await createOrder(
    {
      name: parsed.data.name,
      phone: parsed.data.phone,
      address: parsed.data.address,
      notes: parsed.data.notes,
      items,
      total,
    },
    user.id
  );

  revalidatePath("/orders");
  return { order };
}

export async function cancelOrder(prevState, formData) {
  const orderId = formData.get("orderId");

  const user = await getSession();
  if (!user) return { error: "Not signed in" };

  const order = await getOrder(orderId);
  if (!order) return { error: "Order not found" };

  if (order.userId !== user.id) return { error: "Not yours" };

  if (order.status === "CANCELLED") return { error: "Already cancelled" };

  await markCancelled(orderId);
  revalidatePath("/orders");

  return { ok: true };
}

import { dishes, paginate } from "../../menu/dishes";

export async function GET(request) {
  const { searchParams } = new URL(request.url);
  const q = (searchParams.get("q") ?? "").trim().toLowerCase();

  const matches = q
    ? dishes.filter((dish) =>
        [dish.name, dish.nameAm, dish.category, dish.description, ...dish.ingredients]
          .join(" ")
          .toLowerCase()
          .includes(q)
      )
    : dishes;

  const pageParam = searchParams.get("page");

  if (pageParam !== null) {
    const page = Number(pageParam);
    if (!Number.isInteger(page) || page < 1) {
      return Response.json(
        { error: "page must be a positive whole number" },
        { status: 400 }
      );
    }
    return Response.json(paginate(matches, page));
  }

  return Response.json({ items: matches, total: matches.length });
}

import { orderSchema } from "../../../lib/schema";
import { createOrder, getDish } from "../../../lib/db";
import { getSession } from "../../../lib/session";

export async function POST(request) {
  const user = await getSession();
  if (!user) {
    return Response.json({ error: "Not signed in" }, { status: 401 });
  }

  let body;
  try {
    body = await request.json();
  } catch {
    return Response.json({ error: "Request body must be valid JSON" }, { status: 400 });
  }

  const result = orderSchema.safeParse(body);
  if (!result.success) {
    return Response.json(
      {
        error: "Validation failed",
        fieldErrors: result.error.flatten().fieldErrors,
      },
      { status: 422 }
    );
  }

  if (result.data.dishId) {
    const dish = await getDish(result.data.dishId);
    if (!dish) {
      return Response.json(
        {
          error: "Validation failed",
          fieldErrors: { dishId: ["No such dish"] },
        },
        { status: 422 }
      );
    }
  }

  const order = await createOrder(result.data, user.id);
  return Response.json(order, { status: 201 });
}

"use client";

import Link from "next/link";
import { useSearchParams } from "next/navigation";
import useSWR from "swr";
import { fetcher } from "../../lib/fetcher";
import DishRows from "./DishRows";

export default function PagedDishes({ initial }) {
  const searchParams = useSearchParams();
  const page = Math.max(1, Number(searchParams.get("page")) || 1);

  const { data, error } = useSWR(`/api/dishes?page=${page}`, fetcher, {
    fallbackData: page === 1 ? initial : undefined,
    keepPreviousData: true,
  });

  if (error) {
    return <p role="alert" className="mt-4 text-red-600">Could not load the dishes.</p>;
  }

  if (!data) {
    return <p className="mt-4 text-gray-600">Loading…</p>;
  }

  const stale = data.page !== page;

  return (
    <div>
      <div className={stale ? "opacity-50" : ""}>
        <DishRows items={data.items} />
      </div>

      <nav className="mt-6 flex items-center gap-4">
        {page > 1 ? (
          <Link href={`/dishes?page=${page - 1}`} className="underline">Previous</Link>
        ) : (
          <span className="text-gray-400">Previous</span>
        )}

        <span>Page {data.page} of {data.pageCount}</span>

        {data.page < data.pageCount ? (
          <Link href={`/dishes?page=${page + 1}`} className="underline">Next</Link>
        ) : (
          <span className="text-gray-400">Next</span>
        )}
      </nav>
    </div>
  );
}

function shorten(text, max = 155) {

export async function generateMetadata({ params }) {
  const { id } = await params;
  const dish = await getDish(id);

  if (!dish) {
    return { title: "Dish not found", robots: { index: false } };

    `${dish.name} - ETB ${dish.price}. ${dish.description}`

      description,
      url: `/menu/${dish.id}`,

  isSpecial: item.isSpecial,

const DELAY = 1500;
const wait = (ms) => new Promise((resolve) => setTimeout(resolve, ms));

export async function getCategories() {
  return categories;
}

export async function getDishes() {
  await wait(DELAY);
  return dishes;
}

export async function getDish(id) {
  await wait(DELAY);
  return dishes.find((dish) => dish.id === id);
}
export const categorySlug = (category) =>
  category.toLowerCase().replace(/[^a-z0-9]+/g, "-");

export const PAGE_SIZE = 6;

export function paginate(list, page) {
  const pageCount = Math.max(1, Math.ceil(list.length / PAGE_SIZE));
  const current = Math.min(Math.max(1, page), pageCount);
  const start = (current - 1) * PAGE_SIZE;

  return {
    items: list.slice(start, start + PAGE_SIZE),
    total: list.length,
    page: current,
    pageCount,
  };
}

"use client";

import { useState } from "react";
import Link from "next/link";
import useSWR from "swr";
import { fetcher } from "../../lib/fetcher";
import { useDebounce } from "../../lib/useDebounce";

export default function SearchBox() {
  const [term, setTerm] = useState("");
  const debounced = useDebounce(term.trim(), 300);

  const { data, error, isLoading } = useSWR(
    debounced ? `/api/dishes?q=${encodeURIComponent(debounced)}` : null,
    fetcher,
    { keepPreviousData: true }
  );

  return (
    <div className="mt-4 max-w-md">
      <label className="block font-semibold">Search the menu</label>
      <input
        type="search"
        value={term}
        onChange={(event) => setTerm(event.target.value)}
        placeholder="Try tibs, lentils, fish…"
        className="border rounded px-2 py-1 w-full"
      />

      {debounced && isLoading && <p className="mt-2 text-sm text-gray-600">Searching…</p>}
      {error && <p role="alert" className="mt-2 text-red-600">Search failed.</p>}

      {debounced && data && (
        <div className="mt-2">
          <p className="text-sm text-gray-600">
            {data.total} result{data.total === 1 ? "" : "s"} for "{debounced}"
          </p>
          <ul className="mt-1 space-y-1">
            {data.items.map((dish) => (

"use client";

import { createContext, useCallback, useContext, useEffect, useMemo, useState } from "react";

const CartContext = createContext(null);
const KEY = "mesob_cart";

function isValid(item) {
  return (
    item &&
    typeof item.id === "string" &&
    typeof item.name === "string" &&
    typeof item.price === "number" &&
    Number.isInteger(item.quantity) &&
    item.quantity > 0
  );
}

export function Providers({ children }) {
  const [items, setItems] = useState([]);
  const [loaded, setLoaded] = useState(false);

  useEffect(() => {
    try {
      const saved = JSON.parse(localStorage.getItem(KEY) ?? "[]");
      if (Array.isArray(saved)) setItems(saved.filter(isValid));
    } catch {}
    setLoaded(true);
  }, []);

  useEffect(() => {
    if (!loaded) return;
    try {
      localStorage.setItem(KEY, JSON.stringify(items));
    } catch {}
  }, [items, loaded]);

  const addItem = useCallback((dish) => {
    setItems((current) => {
      const existing = current.find((item) => item.id === dish.id);
      if (existing) {
        return current.map((item) =>
          item.id === dish.id ? { ...item, quantity: item.quantity + 1 } : item
        );
      }
      return [
        ...current,
        { id: dish.id, name: dish.name, price: dish.price, quantity: 1 },
      ];
    });
  }, []);

  const clearCart = useCallback(() => {
    setItems((current) => (current.length ? [] : current));
  }, []);

  const value = useMemo(() => {
    const total = items.reduce((sum, item) => sum + item.price * item.quantity, 0);
    return { items, cart: items, total, addItem, addToCart: addItem, clearCart };
  }, [items, addItem, clearCart]);

  return <CartContext.Provider value={value}>{children}</CartContext.Provider>;
}

export function useCart() {
  return useContext(CartContext);
}

"use server";

import { redirect } from "next/navigation";
import bcrypt from "bcryptjs";
import { registerSchema } from "../../lib/registerSchema";
import { createUser, findUserByPhone } from "../../lib/db";
import { createSession } from "../../lib/session";
import { normalizePhone, safeNext } from "../../lib/authHelpers";

export async function register(prevState, formData) {
  const values = {
    name: String(formData.get("name") ?? ""),
    phone: String(formData.get("phone") ?? ""),
  };
  const next = safeNext(formData.get("next"));

  const parsed = registerSchema.safeParse({
    name: values.name,
    phone: normalizePhone(values.phone),
    password: String(formData.get("password") ?? ""),
    confirm: String(formData.get("confirm") ?? ""),
  });

  if (!parsed.success) {
    return { fieldErrors: parsed.error.flatten().fieldErrors, values };
  }

  const existing = await findUserByPhone(parsed.data.phone);
  if (existing) {
    return {
      fieldErrors: {
        phone: ["An account with this phone number already exists. Sign in instead."],
      },
      values,
    };
  }

  const passwordHash = await bcrypt.hash(parsed.data.password, 10);
  const user = await createUser({
    name: parsed.data.name,
    phone: parsed.data.phone,
    passwordHash,
  });

  await createSession(user);
  redirect(next);
}

import { normalizePhone, safeNext } from "../../lib/authHelpers";

export async function signIn(prevState, formData) {
  const phone = normalizePhone(formData.get("phone"));
  const password = String(formData.get("password") ?? "");
  const next = safeNext(formData.get("next"));

  const user = await findUserByPhone(phone);
  const ok = user && (await bcrypt.compare(password, user.passwordHash));

  if (!ok) {
    return { error: "Phone number or password is incorrect", phone };
  }

  await createSession(user);
  redirect(next);
}

  const p = String(raw ?? "").replace(/[\s-]+/g, "");
  return p.startsWith("+251") ? "0" + p.slice(4) : p;
}

export function safeNext(value) {
  if (typeof value !== "string" || !value.startsWith("/")) return "/";
  try {
    const url = new URL(value, "http://localhost");
    if (url.origin !== "http://localhost") return "/";
    return url.pathname + url.search;
  } catch {
    return "/";
  }
}

"use server";

import "server-only";
import { cookies } from "next/headers";
import { signToken, verifyToken } from "./token";

const COOKIE = "session";

export async function getSession() {
  const token = (await cookies()).get(COOKIE)?.value;
  if (!token) return null;
  const payload = await verifyToken(token);
  if (!payload) return null;
  return { id: payload.id, name: payload.name, role: payload.role };
}

export async function createSession(user) {
  const token = await signToken({ id: user.id, name: user.name, role: user.role });
  (await cookies()).set(COOKIE, token, {
    httpOnly: true,
    secure: process.env.NODE_ENV === "production",
    sameSite: "lax",
    maxAge: 60 * 60 * 24,
    path: "/",
  });
}

export async function destroySession() {
  (await cookies()).delete(COOKIE);
}

import { SignJWT, jwtVerify } from "jose";

const secretValue = process.env.SESSION_SECRET;
if (!secretValue) throw new Error("SESSION_SECRET is not set in .env.local");
const secret = new TextEncoder().encode(secretValue);

export async function signToken(payload) {
  return new SignJWT(payload)
    .setProtectedHeader({ alg: "HS256" })
    .setIssuedAt()
    .setExpirationTime("1d")
    .sign(secret);
}

export async function verifyToken(token) {
  try {
    const { payload } = await jwtVerify(token, secret, { algorithms: ["HS256"] });
    return payload;
  } catch {
    return null;
  }
}

import { useEffect, useState } from "react";

export function useDebounce(value, delay = 300) {
  const [debounced, setDebounced] = useState(value);

  useEffect(() => {
    const timer = setTimeout(() => setDebounced(value), delay);
    return () => clearTimeout(timer);
  }, [value, delay]);

  return debounced;
}

import { NextResponse } from "next/server";
import { verifyToken } from "./lib/token";

export async function proxy(request) {
  const token = request.cookies.get("session")?.value;
  const payload = token ? await verifyToken(token) : null;

  if (!payload) {
    const url = request.nextUrl.clone();
    url.pathname = "/signin";
    url.search = "";
    url.searchParams.set(
      "next",
      request.nextUrl.pathname + request.nextUrl.search
    );
    return NextResponse.redirect(url);
  }

  return NextResponse.next();
}
